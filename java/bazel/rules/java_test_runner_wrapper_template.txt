set -o posix

RUNFILES_MANIFEST_ONLY="%runfiles_manifest_only%"

case "$0" in
  /*) self="$0" ;;
  *)  self="$PWD/$0" ;;
esac

if [[ -z "$RUNFILES_DIR" && -z "$RUNFILES_MANIFEST_FILE" ]]; then
  if [[ -d "${self}.runfiles" ]]; then
    export RUNFILES_DIR="${self}.runfiles"
  elif [[ -f "${self}.runfiles_manifest" ]]; then
    export RUNFILES_MANIFEST_FILE="${self}.runfiles_manifest"
  fi
fi

function rlocation() {
  if [[ -n "$RUNFILES_DIR" ]]; then
    echo "$RUNFILES_DIR/$1"
  elif [[ -n "$RUNFILES_MANIFEST_FILE" ]]; then
    grep -m1 "^$1 " "$RUNFILES_MANIFEST_FILE" | cut -d ' ' -f 2-
  else
    echo "$1"
  fi
}

if [[ -n "$RUNFILES_DIR" ]]; then
  export JAVA_RUNFILES="$RUNFILES_DIR"
fi

TEST_RUNNER=%test_runner%

exec "$TEST_RUNNER" "$@"
